cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2)
include($ENV{PICO_SDK_PATH}/pico_sdk_init.cmake)
project(wsbsd_token C CXX ASM)
pico_sdk_init()
add_executable(wsbsd_token main.c)
target_link_libraries(wsbsd_token pico_stdlib pico_rand pico_unique_id pico_sha256 hardware_otp pico_bootrom)
pico_enable_stdio_usb(wsbsd_token 1)
pico_enable_stdio_uart(wsbsd_token 0)
pico_add_extra_outputs(wsbsd_token)
