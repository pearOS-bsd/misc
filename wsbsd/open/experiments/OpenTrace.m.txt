/* OpenTrace.m — the last messages of the program (what it sent to objects, from its own executable) kept in a ring and printed by the crash handler; OPEN_TRACE=1 also prints them as they go.
 * objc_msgSend is replaced for every image by a dyld interposing tuple: a small assembly trampoline saves the argument registers, asks OpenTraceNote(receiver, selector, return address) and jumps on. */
#import <Foundation/Foundation.h>
#import <objc/runtime.h>
#import <mach-o/dyld.h>
#include <dlfcn.h>
#include <stdint.h>

#define RING 64
static struct { Class cls; SEL sel; } sRing[RING]; static unsigned sPos; static uintptr_t sLo, sHi; static int sOn = -1;
extern void *open_real_msgSend;
void OpenTraceNote(id self, SEL sel, uintptr_t ret)
{
    if (!open_real_msgSend) { void *h = dlopen("/usr/lib/libobjc.A.dylib", RTLD_NOW | RTLD_NOLOAD); open_real_msgSend = h ? dlsym(h, "objc_msgSend") : NULL; }
    if (sOn < 0) {
        sOn = 1; const struct mach_header_64 *mh = (const struct mach_header_64 *)_dyld_get_image_header(0); (void)mh;
        /* text range of the main executable: from its header to the end of the first segment sizes (approximated by the image of address ret at the first call that comes from image 0) */
        uintptr_t base = (uintptr_t)_dyld_get_image_header(0); sLo = base; sHi = base + (1 << 21);
    }
    if (ret < sLo || ret >= sHi || !self) return;
    sRing[sPos % RING].cls = object_getClass(self); sRing[sPos % RING].sel = sel; sPos++;
    if (getenv("OPEN_TRACE")) fprintf(stderr, "open: msg [%s %s]\n", class_getName(object_getClass(self)), sel_getName(sel));
}
void OpenTraceDump(void)
{
    fprintf(stderr, "open: last messages of the program:\n");
    for (unsigned i = sPos > RING ? sPos - RING : 0; i < sPos; i++) fprintf(stderr, "  [%s %s]\n", sRing[i % RING].cls ? class_getName(sRing[i % RING].cls) : "?", sRing[i % RING].sel ? sel_getName(sRing[i % RING].sel) : "?");
}
extern void open_msgSend_trampoline(void);
extern id objc_msgSend(id, SEL, ...);
void *OpenRealMsgSend(void) { return (void *)objc_msgSend; }
__asm__(
".text\n.globl _open_msgSend_trampoline\n_open_msgSend_trampoline:\n"
"  pushq %rbp\n  movq %rsp, %rbp\n"
"  pushq %rdi\n  pushq %rsi\n  pushq %rdx\n  pushq %rcx\n  pushq %r8\n  pushq %r9\n  pushq %rax\n  pushq %rax\n"	/* 8 pushes: 64 bytes, keeps the stack 16 aligned */
"  subq $128, %rsp\n  movdqu %xmm0, 0(%rsp)\n  movdqu %xmm1, 16(%rsp)\n  movdqu %xmm2, 32(%rsp)\n  movdqu %xmm3, 48(%rsp)\n  movdqu %xmm4, 64(%rsp)\n  movdqu %xmm5, 80(%rsp)\n  movdqu %xmm6, 96(%rsp)\n  movdqu %xmm7, 112(%rsp)\n"
"  movq 8(%rbp), %rdx\n"								/* the return address of the caller */
"  call _OpenTraceNote\n"
"  movdqu 0(%rsp), %xmm0\n  movdqu 16(%rsp), %xmm1\n  movdqu 32(%rsp), %xmm2\n  movdqu 48(%rsp), %xmm3\n  movdqu 64(%rsp), %xmm4\n  movdqu 80(%rsp), %xmm5\n  movdqu 96(%rsp), %xmm6\n  movdqu 112(%rsp), %xmm7\n  addq $128, %rsp\n"
"  popq %rax\n  popq %rax\n  popq %r9\n  popq %r8\n  popq %rcx\n  popq %rdx\n  popq %rsi\n  popq %rdi\n"
"  popq %rbp\n"
"  jmp *_open_real_msgSend(%rip)\n"
".data\n.globl _open_real_msgSend\n_open_real_msgSend: .quad 0\n"
);
__attribute__((used)) static struct { const void *r, *o; } sTraceInterpose[] __attribute__((section("__DATA,__interpose"))) = { { (const void *)open_msgSend_trampoline, (const void *)objc_msgSend } };
