/* Note: NSTextRange/NSTextLocation live in UIFoundation (re-exported by AppKit),
   not the AppKit image. Measured semantics (AppKit runtime, macOS 27 26A5416b):
   isEmpty = location==endLocation; containsLocation: start<=loc<end;
   intersects = positive-length overlap; union = [min start,max end];
   intersection = [max start,min end) or nil when disjoint;
   containsRange: self.start<=other.start && other.end<=self.end. */
